  sample(c(-1, 0, 1), size = n, replace = TRUE, prob = c(theta / 2, theta / 3, 1 - 5 * theta / 6))

Ejercicio 5

Código 
set.seed(123)
theta_fijo5 <- 5 # Valor de prueba para theta

# 1. Función de densidad
dexr5 <- function(x, theta) {
  ifelse(x >= 0 & x <= theta, (2 * x) / (theta^2), 0)
}

# 2. Generación de muestra aleatoria por inversión: X = theta * sqrt(U)
rexr5 <- function(n, theta) {
  U <- runif(n)
  theta * sqrt(U)
}

# 3. Estimador por el método de momentos: hat(theta) = 3 * bar(X) / 2
estimar_theta5 <- function(X) {
  (3 * mean(X)) / 2
}

# Simulación con n = 1000
muestra5 <- rexr5(1000, theta_fijo5)
theta_hat5 <- estimar_theta5(muestra5)
cat("Estimación de theta (Ejercicio 5):", theta_hat5, "\n")
